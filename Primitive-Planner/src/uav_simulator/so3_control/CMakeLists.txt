cmake_minimum_required(VERSION 3.8)
project(so3_control)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(uav_utils REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
include_directories(include ${EIGEN3_INCLUDE_DIRS})
add_library(SO3Control src/SO3Control.cpp)
add_executable(so3_control_ros2 src/so3_control_ros2.cpp)
add_executable(control_example src/control_example.cpp)
foreach(t SO3Control control_example so3_control_ros2)
  ament_target_dependencies(${t} rclcpp nav_msgs quadrotor_msgs uav_utils sensor_msgs std_msgs)
endforeach()
target_link_libraries(control_example SO3Control)
target_link_libraries(so3_control_ros2 SO3Control)
install(TARGETS SO3Control control_example so3_control_ros2 ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
