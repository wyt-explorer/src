cmake_minimum_required(VERSION 3.8)
project(local_sensing_node)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(pcl_render_node src/pointcloud_render_node.cpp)
target_include_directories(pcl_render_node PRIVATE ${PCL_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
target_link_libraries(pcl_render_node ${PCL_LIBRARIES})
ament_target_dependencies(pcl_render_node rclcpp sensor_msgs geometry_msgs nav_msgs quadrotor_msgs pcl_conversions)
install(TARGETS pcl_render_node DESTINATION lib/${PROJECT_NAME})
ament_package()
