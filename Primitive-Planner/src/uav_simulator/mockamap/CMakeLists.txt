cmake_minimum_required(VERSION 3.8)
project(mockamap)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(pcl_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED)
find_package(Eigen3 REQUIRED)
file(GLOB mockamap_SRCS src/*.cpp)
add_executable(mockamap_node ${mockamap_SRCS})
target_include_directories(mockamap_node PRIVATE include ${PCL_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
target_link_libraries(mockamap_node ${PCL_LIBRARIES})
ament_target_dependencies(mockamap_node rclcpp pcl_ros pcl_conversions)
install(TARGETS mockamap_node DESTINATION lib/${PROJECT_NAME})
ament_package()
