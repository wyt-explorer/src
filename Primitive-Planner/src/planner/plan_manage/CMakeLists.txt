cmake_minimum_required(VERSION 3.8)
project(primitive_planner)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(traj_utils REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED)
find_package(Eigen3 REQUIRED)
include_directories(include ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_executable(primitive_planner_node src/primitive_planner_node.cpp src/pp_replan_fsm.cpp src/planner_manager.cpp)
add_executable(traj_server src/traj_server.cpp)
foreach(t primitive_planner_node traj_server)
  target_link_libraries(${t} ${PCL_LIBRARIES})
  ament_target_dependencies(${t} rclcpp std_msgs geometry_msgs nav_msgs quadrotor_msgs traj_utils pcl_conversions)
endforeach()
install(TARGETS primitive_planner_node traj_server DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch primitive_library DESTINATION share/${PROJECT_NAME})
install(PROGRAMS
  ../../scripts/gen_position_swap.py ../../scripts/pub_trigger.sh ../../scripts/run_swarm.sh
  DESTINATION lib/${PROJECT_NAME})
ament_package()
