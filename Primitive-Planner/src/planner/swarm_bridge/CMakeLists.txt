cmake_minimum_required(VERSION 3.8)
project(swarm_bridge)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(traj_utils REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
add_executable(bridge_node_udp src/bridge_node_udp.cpp)
target_include_directories(bridge_node_udp PRIVATE include ${EIGEN3_INCLUDE_DIRS})
ament_target_dependencies(bridge_node_udp rclcpp std_msgs geometry_msgs sensor_msgs traj_utils)
target_link_libraries(bridge_node_udp Threads::Threads)
install(TARGETS bridge_node_udp DESTINATION lib/${PROJECT_NAME})
ament_package()
