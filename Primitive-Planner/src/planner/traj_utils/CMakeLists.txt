cmake_minimum_required(VERSION 3.8)
project(traj_utils)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(std_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Eigen3 REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  msg/SwarmPrimitiveTraj.msg msg/Polynomial.msg
  DEPENDENCIES builtin_interfaces std_msgs geometry_msgs visualization_msgs)
add_library(traj_visualization src/planning_visualization.cpp)
target_include_directories(traj_visualization PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include/${PROJECT_NAME}>)
target_link_libraries(traj_visualization Eigen3::Eigen)
ament_target_dependencies(traj_visualization rclcpp geometry_msgs visualization_msgs std_msgs)
install(TARGETS traj_visualization EXPORT export_traj_visualization ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime rclcpp geometry_msgs visualization_msgs std_msgs Eigen3)
ament_export_include_directories(include/${PROJECT_NAME})
ament_export_targets(export_traj_visualization HAS_LIBRARY_TARGET)
ament_package()
