cmake_minimum_required(VERSION 3.8)
project(random_goals)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(random_goals_node src/random_goals_node.cpp)
target_include_directories(random_goals_node PRIVATE ${EIGEN3_INCLUDE_DIRS})
ament_target_dependencies(random_goals_node rclcpp nav_msgs quadrotor_msgs)
install(TARGETS random_goals_node DESTINATION lib/${PROJECT_NAME})
ament_package()
