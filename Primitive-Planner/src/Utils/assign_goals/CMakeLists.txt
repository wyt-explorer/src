cmake_minimum_required(VERSION 3.8)
project(assign_goals)
find_package(ament_cmake REQUIRED)
find_package(uav_utils REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(assign_goals_node src/assign_goals_node.cpp)
target_include_directories(assign_goals_node PRIVATE ${EIGEN3_INCLUDE_DIRS})
ament_target_dependencies(assign_goals_node rclcpp geometry_msgs visualization_msgs quadrotor_msgs uav_utils)
install(TARGETS assign_goals_node DESTINATION lib/${PROJECT_NAME})
ament_package()
