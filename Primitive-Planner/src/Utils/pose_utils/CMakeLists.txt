cmake_minimum_required(VERSION 3.8)
project(pose_utils)
find_package(ament_cmake REQUIRED)
find_package(Armadillo REQUIRED)
add_library(pose_utils src/pose_utils.cpp)
target_include_directories(pose_utils PUBLIC include ${ARMADILLO_INCLUDE_DIRS})
target_link_libraries(pose_utils ${ARMADILLO_LIBRARIES})
install(TARGETS pose_utils ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_libraries(pose_utils)
ament_package()
